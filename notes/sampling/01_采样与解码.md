```python
def softmax(z):                                    # (V,) → (V,)
    e = np.exp(z - z.max())                        # 先减最大值防溢出
    return e / e.sum()

def top_k_filter(logits, k):                       # (V,) → (V,)
    kth = np.sort(logits)[-k]                      # 第 k 大的值
    return np.where(logits < kth, -np.inf, logits)

def top_p_filter(probs, p):                        # (V,) → (V,)
    order = np.argsort(-probs)                     # 概率从大到小的 token id
    sorted_probs = probs[order]
    cum_before = np.cumsum(sorted_probs) - sorted_probs   # 每个 token「之前」的累计概率
    out = probs.copy()
    out[order[cum_before > p]] = 0.0               # 之前已够 p → 不需要它；最大的永远保留
    return out / out.sum()

def sample(logits, k=0, p=1.0, temperature=1.0, rng=np.random.default_rng()):
    z = logits / temperature                       # 1. 温度
    if k > 0: z = top_k_filter(z, k)               # 2. top-k
    probs = softmax(z)                             # 3. 归一化
    if p < 1.0: probs = top_p_filter(probs, p)     # 4. top-p
    return int(rng.choice(len(probs), p=probs))    # 5. 抽样
```

- **Temperature**：<1 更尖锐，>1 更平坦，→0 即 greedy。
- **top-p 细节**：用 `csum - probs > p` 剔除，保证至少保留 1 个 token（最高概率那个）。
- 加权随机采样（前缀和 + 二分）：`cs = np.cumsum(w); i = np.searchsorted(cs, rng.random()*cs[-1], side='right')`。
- **Beam Search**：每步对 `beam×V` 个候选取 top-beam（log 概率累加）；长度归一化 `score/len^α`；遇 EOS 的 beam 移入完成列表。

```python
from typing import List, Optional, Tuple

import numpy as np


def softmax(z: np.ndarray) -> np.ndarray:
    """z: (V,) float -> (V,) probabilities. Subtract the max first so exp never overflows."""
    e = np.exp(z - z.max())
    return e / e.sum()


def top_k_filter(logits: np.ndarray, k: int) -> np.ndarray:
    """Keep the k largest logits, set the rest to -inf. logits: (V,) -> (V,)."""
    kth_largest = np.sort(logits)[-k]                           # value of the k-th largest
    return np.where(logits < kth_largest, -np.inf, logits)


def top_p_filter(probs: np.ndarray, p: float) -> np.ndarray:
    """Nucleus filter: keep the smallest set of tokens whose mass >= p, renormalise. probs: (V,) -> (V,)."""
    order = np.argsort(-probs)                                  # token ids, most likely first
    sorted_probs = probs[order]                                 # (V,)
    cum_before = np.cumsum(sorted_probs) - sorted_probs         # mass BEFORE each token
    remove = cum_before > p                                     # token not needed to reach p
    out = probs.copy()
    out[order[remove]] = 0.0                                    # the top token is never removed
    return out / out.sum()


def sample_top_k_top_p(logits: np.ndarray, k: int = 0, p: float = 1.0, temperature: float = 1.0,
                       rng: Optional[np.random.Generator] = None) -> int:
    """Sample one token id: temperature -> top-k -> softmax -> top-p -> draw.

    Args:
        logits: (V,) float raw scores.
        k: keep the top-k tokens (0 = disabled).
        p: nucleus mass in (0, 1] (1.0 = disabled).
        temperature: > 0; lower = sharper.
        rng: optional numpy Generator for reproducibility.
    Returns:
        Sampled token id in [0, V).
    """
    rng = rng or np.random.default_rng()
    z = logits / temperature                                    # (V,)
    if k > 0:
        z = top_k_filter(z, k)
    probs = softmax(z)                                          # (V,)
    if p < 1.0:
        probs = top_p_filter(probs, p)
    return int(rng.choice(len(probs), p=probs))
```
